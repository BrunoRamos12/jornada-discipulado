async function enviarEntrada() { const btn = document.getElementById("btnEnviar"); btn.disabled = true; btn.textContent = "Enviando..."; try { const payload = { dificuldade: document.querySelector('input[name="dificuldade"]:checked')?.value || "", situacao: document.querySelector('input[name="situacao"]:checked')?.value || "", diario: document.getElementById("diario")?.value || "", pedido_oracao: document.getElementById("oracao")?.value || "", extra: Array.from( document.querySelectorAll('input[name="extra"]:checked') ).map(e => e.value) }; const r = await fetch("/api/entrada", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (!r.ok) { throw new Error("Erro ao enviar"); } // ==================================== // MENSAGEM DE SUCESSO // ==================================== const ok = document.getElementById("okForm"); if (ok) { ok.style.display = "block"; ok.innerHTML = "Mensagem enviada ao pastor com sucesso!"; } // ==================================== // LIMPA RADIO // ==================================== document .querySelectorAll('input[type="radio"]') .forEach(r => { r.checked = false; }); // ==================================== // LIMPA CHECKBOX // ==================================== document .querySelectorAll('input[type="checkbox"]') .forEach(c => { c.checked = false; }); // ==================================== // LIMPA CAMPOS TEXTO // ==================================== const diario = document.getElementById("diario"); if (diario) { diario.value = ""; } const oracao = document.getElementById("oracao"); if (oracao) { oracao.value = ""; } const extra = document.getElementById("extra"); if (extra) { extra.value = ""; } // ==================================== // RESTAURA BOTÃO // ==================================== btn.disabled = false; btn.textContent = "Enviar ao pastor"; // ==================================== // VOLTA TOPO // ==================================== window.scrollTo({ top: 0, behavior: "smooth" }); } catch (e) { const err = document.getElementById("errForm"); if (err) { err.style.display = "block"; err.innerHTML = "Não foi possível enviar sua entrada. Tente novamente."; } btn.disabled = false; btn.textContent = "Enviar ao pastor"; console.error(e); } }

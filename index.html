async function enviarEntrada() { const btn = document.getElementById("btnEnviar"); btn.disabled = true; btn.textContent = "Enviando..."; try { const payload = { dificuldade: document.querySelector('input[name="dificuldade"]:checked')?.value || "", situacao: document.querySelector('input[name="situacao"]:checked')?.value || "", diario: document.getElementById("diario").value, pedido_oracao: document.getElementById("oracao").value, extra: Array.from( document.querySelectorAll('input[name="extra"]:checked') ).map(e => e.value) }; const r = await fetch("/api/entrada", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(payload) }); if (!r.ok) { throw new Error("Erro ao enviar"); } // ========================= // SUCESSO // ========================= showOk( "okForm", "Mensagem enviada ao pastor com sucesso!" ); // limpa radios document .querySelectorAll('input[type="radio"]') .forEach(r => r.checked = false); // limpa checkboxes document .querySelectorAll('input[type="checkbox"]') .forEach(c => c.checked = false); // limpa campos texto document.getElementById("diario").value = ""; document.getElementById("oracao").value = ""; // limpa extra se existir const extra = document.getElementById("extra"); if (extra) { extra.value = ""; } // reativa botão btn.disabled = false; btn.textContent = "Enviar ao pastor"; // sobe a página window.scrollTo({ top: 0, behavior: "smooth" }); } catch (e) { showErr( "errForm", "Não foi possível enviar sua entrada. Tente novamente." ); btn.disabled = false; btn.textContent = "Enviar ao pastor"; console.error(e); } }
